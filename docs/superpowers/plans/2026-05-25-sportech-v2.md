# Sportech v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a cinematic 9-section snap-scroll landing page for Sportech with GSAP animations, gpt-image-2 AI backgrounds, and an interactive revenue simulator.

**Architecture:** Clean slate on the `v2` branch of the existing Sportech repo. All components, layouts, and styles are rebuilt from scratch; only `src/data/clubs.ts` (702 clubs) and the revenue simulator config are ported. CSS `scroll-snap` handles section snapping; `IntersectionObserver` (via `useInView`) triggers GSAP timelines on section entry; Framer Motion handles interactive micro-animations. Four background images are generated once via `scripts/generate-images.ts` and committed to `public/images/v2/`.

**Tech Stack:** Next.js 16.2.4 (App Router, already installed), React 19 (already installed), TypeScript strict, Tailwind CSS v4 (already installed), GSAP 3 (new), Framer Motion 12 (already installed), OpenAI SDK (dev-only, for image gen script), Vitest + Testing Library (new)

```typescript
import { useEffect, useRef, useState } from 'react'

export function useInView<T extends Element = HTMLElement>(threshold = 0.35) {
  const ref = useRef<T>(null)
  const [inView, setInView] = useState(false)

  useEffect(() => {
    const el = ref.current
    if (!el) return

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setInView(true)
          observer.disconnect() // fire once only
        }
      },
      { threshold }
    )
    observer.observe(el)
    return () => observer.disconnect()
  }, [threshold])

  return { ref, inView }
}
```

```typescript
import type { StreamKey } from '@/data/simulator'

const CONVERSION_RATE = 0.25
const SHARE_FACTOR = 0.5

export function calculateStreamRevenue(fans: number, gross: number): number {
  return Math.round(fans * CONVERSION_RATE * gross * SHARE_FACTOR)
}

export function calculateTotalRevenue(fans: number, activeStreams: Set<string>): number {
  return STREAMS
    .filter(s => activeStreams.has(s.key))
    .reduce((sum, s) => sum + calculateStreamRevenue(fans, s.gross), 0)
}
```

```typescript
import OpenAI from 'openai'
import * as fs from 'fs/promises'
import * as path from 'path'
import * as dotenv from 'dotenv'

dotenv.config({ path: path.join(process.cwd(), '.env.local') })

const apiKey = process.env.OPENAI_API_KEY
if (!apiKey) throw new Error('OPENAI_API_KEY not set in .env.local')

const client = new OpenAI({ apiKey })

// 1920×1088: both divisible by 16, ~16:9. gpt-image-2 requires divisible-by-16 dimensions.
const SIZE = '1920x1088'

const IMAGES = [
  {
    name: 'hero',
    prompt:
      'Cinematic aerial view of a packed football stadium at night, seen directly from above. Deep black sky, vivid crowd glow in yellow and white light, neon yellow holographic data overlays floating above the crowd showing revenue figures. Ultra-dark moody atmosphere, photorealistic, 4K.',
  },
  {
    name: 'problem',
    prompt:
      'Wide aerial view of a massive sports stadium at night zooming out to a dark city skyline. Glowing cyan and green revenue numbers and data streams float above the crowd like digital holograms. Billions of dollars visualised as light particles above the fans. Dark cinematic mood, photorealistic, 4K.',
  },
  {
    name: 'scope',
    prompt:
      'Eight professional athletes from different sports — soccer, basketball, cricket, tennis, esports, rugby, boxing, and athletics — in dramatic action poses arranged symmetrically. Dark studio background with neon yellow rim lighting and cinematic fog. Photorealistic, 4K.',
  },
  {
    name: 'future',
    prompt:
      'Futuristic sports fan experience: holographic AR overlays projected on a stadium, gaming controller interfaces, glowing NFT digital trading cards, live streaming panels, social media feeds, metaverse fan avatars. Dark sci-fi atmosphere, yellow and purple neon lighting, highly detailed, 4K cinematic.',
  },
]

async function main() {
  const outDir = path.join(process.cwd(), 'public', 'images', 'v2')
  await fs.mkdir(outDir, { recursive: true })

  for (const img of IMAGES) {
    console.log(`Generating ${img.name}...`)
    const response = await client.images.generate({
      model: 'gpt-image-2',
      prompt: img.prompt,
      size: SIZE as never, // gpt-image-2 accepts arbitrary sizes not in the SDK enum
      quality: 'high',
      n: 1,
      response_format: 'b64_json',
    })

    const b64 = response.data[0].b64_json
    if (!b64) throw new Error(`No image data returned for ${img.name}`)

    const buffer = Buffer.from(b64, 'base64')
    const filePath = path.join(outDir, `${img.name}.png`)
    await fs.writeFile(filePath, buffer)
    console.log(`  ✓ Saved public/images/v2/${img.name}.png`)
  }

  console.log('\nAll 4 images generated successfully.')
}

main().catch(err => {
  console.error(err.message)
  process.exit(1)
})
```

```tsx
'use client'
import { useEffect, useRef } from 'react'
import { gsap } from '@/lib/gsap'

interface AnimatedCounterProps {
  value: number
  prefix?: string
  suffix?: string
  duration?: number
  decimals?: number
  className?: string
  style?: React.CSSProperties
  trigger: boolean
}

export function AnimatedCounter({
  value, prefix = '', suffix = '', duration = 2.2,
  decimals = 0, className, style, trigger,
}: AnimatedCounterProps) {
  const nodeRef = useRef<HTMLSpanElement>(null)
  const obj = useRef({ val: 0 })

  useEffect(() => {
    if (!trigger || !nodeRef.current) return
    const node = nodeRef.current
    gsap.to(obj.current, {
      val: value,
      duration,
      ease: 'power2.out',
      onUpdate() {
        node.textContent =
          prefix + obj.current.val.toFixed(decimals).replace(/\B(?=(\d{3})+(?!\d))/g, ',') + suffix
      },
    })
  }, [trigger, value, prefix, suffix, duration, decimals])

  return (
    <span ref={nodeRef} className={className} style={style}>
      {prefix}0{suffix}
    </span>
  )
}
```

```tsx
function polar(cx: number, cy: number, r: number, angleDeg: number) {
  const rad = ((angleDeg - 90) * Math.PI) / 180
  return { x: cx + r * Math.cos(rad), y: cy + r * Math.sin(rad) }
}

export function Streams() {
  const { ref, inView } = useInView<HTMLElement>()
  const svgRef = useRef<SVGSVGElement>(null)

  useEffect(() => {
    if (!inView || !svgRef.current) return
    const svg = svgRef.current

    const center = svg.querySelector<SVGGElement>('.center-grp')

    gsap.set(spokes, { strokeDashoffset: 200, opacity: 0 })
    gsap.set(nodes, { scale: 0, opacity: 0, transformOrigin: 'center center' })
    gsap.set(labels, { opacity: 0, y: 8 })

    const tl = gsap.timeline()
    if (center) tl.to(center, { scale: 1, opacity: 1, duration: 0.5, ease: 'back.out(1.5)' })
```

```tsx
  const angle = (i / 40) * 2 * Math.PI
  const layer = Math.floor(i / 10)
  const r = 80 + layer * 25
  return {
    id: i,
    x: 200 + r * Math.cos(angle),
    y: 200 + r * Math.sin(angle) * 0.4, // flatten to ellipse
    delay: `${(i * 0.07) % 2}s`,
  }

export function CTA() {
  const { ref, inView } = useInView<HTMLElement>(0.1)
  const contentRef = useRef<HTMLDivElement>(null)
  const globeRef = useRef<SVGSVGElement>(null)

  useEffect(() => {
    if (!inView) return
    const tl = gsap.timeline()
    if (globeRef.current) {

      tl.to(globeRef.current, { scale: 1, opacity: 1, duration: 1.2, ease: 'power2.out' })
    }
    if (contentRef.current) {
```

```tsx
'use client'
import { useEffect, useState } from 'react'

const SECTIONS = ['hero', 'problem', 'scope', 'streams', 'brand', 'simulator', 'infra', 'future', 'cta']

export function NavDots() {
  const [active, setActive] = useState(0)

  useEffect(() => {
    const container = document.getElementById('snap-container')
    if (!container) return

    function onScroll() {
      const index = Math.round(container.scrollTop / window.innerHeight)
      setActive(Math.min(Math.max(index, 0), SECTIONS.length - 1))
    }

    container.addEventListener('scroll', onScroll, { passive: true })
    return () => container.removeEventListener('scroll', onScroll)
  }, [])

  function scrollTo(index: number) {
    const container = document.getElementById('snap-container')
    container?.scrollTo({ top: index * window.innerHeight, behavior: 'smooth' })
  }

  return (
    <div style={{
      position: 'fixed', right: 20, top: '50%', transform: 'translateY(-50%)',
      zIndex: 100, display: 'flex', flexDirection: 'column', gap: 8,
      pointerEvents: 'auto',
    }}>
      {SECTIONS.map((id, i) => (
        <button
          key={id}
          onClick={() => scrollTo(i)}
          aria-label={`Go to ${id} section`}
          style={{
            width: active === i ? 8 : 5,
            height: active === i ? 8 : 5,
            borderRadius: '50%',
            background: active === i ? '#ffe600' : '#333',
            border: 'none', padding: 0, cursor: 'pointer',
            transition: 'all 0.25s ease',
            display: 'block',
          }}
        />
      ))}
    </div>
  )
}
```
